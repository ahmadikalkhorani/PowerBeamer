\begin{frame}{Step 1: Annotation \& Inking Suite}
  \begin{block}{Live Pen \& Highlighter Tools}
    Annotate directly on slides with sub-10ms Catmull-Rom smoothed vector ink:
    \begin{itemize}
      \item \textbf{Pen (\texttt{P})}: Choose from Fountain, Ballpoint, or Brush tips.
      \item \textbf{Highlighter (\texttt{H})}: Uses multiply blend mode to preserve sharp text.
      \item \textbf{Eraser (\texttt{E})}: Fast stroke-collision eraser.
      \item \textbf{Laser Pointer (\texttt{L})}: Glowing trail mirrored live to your audience.
    \end{itemize}
  \end{block}

  \begin{exampleblock}{Derivation Sandbox}
    Practice drawing or writing formulas below:
    \begin{equation}
      \mathcal{L}(\theta) = \sum_{i=1}^N \log p(y_i \mid x_i; \theta) - \frac{\lambda}{2}\|\theta\|_2^2
    \end{equation}
  \end{exampleblock}

  \note{
    \textbf{SPEAKER NOTES (Slide 2)}:
    \begin{itemize}
      \item Switch to the \textbf{Pen tool (P)} and circle the regularization term $\frac{\lambda}{2}\|\theta\|_2^2$.
      \item Try changing ink colors using the color palette in the top bar.
      \item Press \textbf{Cmd+Z} / \textbf{Ctrl+Z} to test multi-level undo.
      \item Press \textbf{N} or \textbf{+} to spawn a blank whiteboard whenever you need extra writing space!
    \end{itemize}
  }
\end{frame}
